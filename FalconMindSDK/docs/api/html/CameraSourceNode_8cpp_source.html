60 bool CameraSourceNode::initV4L2() {
61  if (config_.device.empty()) return false;
62  v4l2Fd_ = open(config_.device.c_str(), O_RDWR);
63  if (v4l2Fd_ < 0) {
64  std::cerr << "[CameraSourceNode] open " << config_.device << " failed: " << errno << std::endl;
65  return false;
66  }
67  v4l2_capability cap{};
68  if (ioctl(v4l2Fd_, VIDIOC_QUERYCAP, &cap) != 0 ||
69  !(cap.capabilities & V4L2_CAP_VIDEO_CAPTURE) ||
70  !(cap.capabilities & V4L2_CAP_STREAMING)) {
71  std::cerr << "[CameraSourceNode] device not a capture/streaming device" << std::endl;
72  close(v4l2Fd_);
73  v4l2Fd_ = -1;
74  return false;
75  }
76  int w = config_.width > 0 ? static_cast<int>(config_.width) : 640;
77  int h = config_.height > 0 ? static_cast<int>(config_.height) : 480;
78  uint32_t pixFmt = 0;
79  if (!trySetFormat(v4l2Fd_, w, h, &pixFmt, &v4l2Stride_)) {
80  std::cerr << "[CameraSourceNode] SET_FMT failed" << std::endl;
81  close(v4l2Fd_);
82  v4l2Fd_ = -1;
83  return false;
84  }
85  v4l2Width_ = w;
86  v4l2Height_ = h;
87  if (v4l2Stride_ <= 0) v4l2Stride_ = (pixFmt == V4L2_PIX_FMT_RGB24) ? (w * 3) : (w * 2);
88 
89  v4l2_requestbuffers req{};
90  req.count = 4;
91  req.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
92  req.memory = V4L2_MEMORY_MMAP;
93  if (ioctl(v4l2Fd_, VIDIOC_REQBUFS, &req) != 0 || req.count == 0) {
94  std::cerr << "[CameraSourceNode] REQBUFS failed" << std::endl;
95  close(v4l2Fd_);
96  v4l2Fd_ = -1;
97  return false;
98  }
99  v4l2NumBuffers_ = req.count;
100  v4l2MapPtrs_.resize(v4l2NumBuffers_, nullptr);
101  v4l2MapLens_.resize(v4l2NumBuffers_, 0);
102  for (unsigned i = 0; i < v4l2NumBuffers_; ++i) {
103  v4l2_buffer buf{};
104  buf.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
105  buf.memory = V4L2_MEMORY_MMAP;
106  buf.index = i;
107  if (ioctl(v4l2Fd_, VIDIOC_QUERYBUF, &buf) != 0) {
108  shutdownV4L2();
109  return false;
110  }
111  void* ptr = mmap(nullptr, buf.length, PROT_READ | PROT_WRITE, MAP_SHARED, v4l2Fd_, buf.m.offset);
112  if (ptr == MAP_FAILED) {
113  shutdownV4L2();
114  return false;
115  }
116  v4l2MapPtrs_[i] = ptr;
117  v4l2MapLens_[i] = buf.length;
118  }
119  for (unsigned i = 0; i < v4l2NumBuffers_; ++i) {
120  v4l2_buffer buf{};
121  buf.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
122  buf.memory = V4L2_MEMORY_MMAP;
123  buf.index = i;
124  if (ioctl(v4l2Fd_, VIDIOC_QBUF, &buf) != 0) {
125  shutdownV4L2();
126  return false;
127  }
128  }
129  enum v4l2_buf_type type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
130  if (ioctl(v4l2Fd_, VIDIOC_STREAMON, &type) != 0) {
131  shutdownV4L2();
132  return false;
133  }
134 
135  size_t pixelSize = static_cast<size_t>(v4l2Stride_) * static_cast<size_t>(v4l2Height_);
136  size_t rgbSize = static_cast<size_t>(v4l2Width_) * static_cast<size_t>(v4l2Height_) * 3u;
137  frameBuffer_.resize(sizeof(CameraFramePacket) + std::max(pixelSize, rgbSize));
138  CameraFramePacket* header = reinterpret_cast<CameraFramePacket*>(frameBuffer_.data());
139  header->width = v4l2Width_;
140  header->height = v4l2Height_;
141  header->stride = v4l2Stride_;
142  std::strncpy(header->format, (pixFmt == V4L2_PIX_FMT_RGB24) ? "RGB8" : "YUYV", sizeof(header->format) - 1);
143  header->format[sizeof(header->format) - 1] = '\0';
144  return true;
145 }

227 void CameraSourceNode::process() {
228  if (!started_) return;
229 
230 #ifdef __linux__
231  if (v4l2Ready_ && v4l2Fd_ >= 0) {
232  v4l2_buffer buf{};
233  buf.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
234  buf.memory = V4L2_MEMORY_MMAP;
235  if (ioctl(v4l2Fd_, VIDIOC_DQBUF, &buf) != 0)
236  return;
237  CameraFramePacket* header = reinterpret_cast<CameraFramePacket*>(frameBuffer_.data());
238  std::uint8_t* dst = frameBuffer_.data() + sizeof(CameraFramePacket);
239  const std::uint8_t* src = static_cast<const std::uint8_t*>(v4l2MapPtrs_[buf.index]);
240  size_t srcStride = static_cast<size_t>(v4l2Stride_);
241  int w = v4l2Width_, hh = v4l2Height_;
242  bool isYuyv = (std::strcmp(header->format, "YUYV") == 0);
243  if (isYuyv && srcStride >= static_cast<size_t>(w * 2)) {
244  for (int y = 0; y < hh; ++y) {
245  const std::uint8_t* row = src + y * srcStride;
246  for (int x = 0; x < w; ++x) {
247  int u = row[x << 1];
248  int y0 = row[x << 1 | 1];
249  int v = (x + 1) < w ? row[(x + 1) << 1] : u;
250  int r = y0 + (351 * (v - 128)) / 256;
251  int g = y0 - (179 * (v - 128) + 86 * (u - 128)) / 256;
252  int b = y0 + (443 * (u - 128)) / 256;
253  dst[(y * w + x) * 3 + 0] = static_cast<std::uint8_t>(std::max(0, std::min(255, r)));
254  dst[(y * w + x) * 3 + 1] = static_cast<std::uint8_t>(std::max(0, std::min(255, g)));
255  dst[(y * w + x) * 3 + 2] = static_cast<std::uint8_t>(std::max(0, std::min(255, b)));
256  }
257  }
258  header->stride = w * 3;
259  std::strncpy(header->format, "RGB8", sizeof(header->format) - 1);
260  header->format[sizeof(header->format) - 1] = '\0';
261  } else {
262  size_t pixelSize = srcStride * static_cast<size_t>(hh);
263  std::memcpy(dst, src, std::min(pixelSize, v4l2MapLens_[buf.index]));
264  }
265  size_t pixelSize = static_cast<size_t>(header->stride) * static_cast<size_t>(header->height);
266  ioctl(v4l2Fd_, VIDIOC_QBUF, &buf);
267 
268  auto pad = getPad("video_out");
269  if (pad)
270  pad->pushToConnections(frameBuffer_.data(), sizeof(CameraFramePacket) + pixelSize);
271  return;
272  }
273 #endif
274 
275  if (fileMode_ && fileStream_.is_open()) {
276  std::uint8_t* dst = frameBuffer_.data() + sizeof(CameraFramePacket);
277  fileStream_.read(reinterpret_cast<char*>(dst), static_cast<std::streamsize>(fileFrameBytes_));
278  if (fileStream_.gcount() == static_cast<std::streamsize>(fileFrameBytes_)) {
279  auto pad = getPad("video_out");
280  if (pad)
281  pad->pushToConnections(frameBuffer_.data(), sizeof(CameraFramePacket) + fileFrameBytes_);
282  }
283  if (fileStream_.eof()) {
284  fileStream_.clear();
285  fileStream_.seekg(0);
286  }
287  return;
288  }
