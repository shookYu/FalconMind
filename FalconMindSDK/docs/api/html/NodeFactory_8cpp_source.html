32 std::shared_ptr<Node> NodeFactory::createNode(const std::string& template_id,
33  const std::string& node_id,
34  const void* params) {
35  // 阶段4优化：确保已初始化（线程安全）
36  if (!initialized_.load(std::memory_order_acquire)) {
37  std::lock_guard<std::mutex> lock(init_mutex_);
38  if (!initialized_.load(std::memory_order_relaxed)) {
39  initializeDefaultTypes();
40  initialized_.store(true, std::memory_order_release);
41  }
42  }
43 
44  // 阶段4优化：使用find而不是count，减少一次查找
45  // unordered_map::find是O(1)，已经很快
46  // 阶段4优化：可以考虑使用string_view减少字符串拷贝（C++17）
47  // 但当前性能已经很好，暂时不引入C++17依赖
48  auto it = creators_.find(template_id);
49  if (it == creators_.end()) {
50  std::cerr << "NodeFactory: Unknown template_id: " << template_id << std::endl;
51  return nullptr;
52  }
53 
54  try {
55  // 直接调用创建函数，避免额外的查找
56  // 阶段4优化：对于频繁创建的同类型节点，可以考虑缓存创建函数指针
57  // 但当前性能已经很好（0.001ms/节点），暂时不需要进一步优化
58  return it->second(node_id, params);
59  } catch (const std::exception& e) {
60  std::cerr << "NodeFactory: Failed to create node " << template_id
61  << ": " << e.what() << std::endl;
62  return nullptr;
63  }
