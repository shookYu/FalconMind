26 static bool isPointInPolygon(const GeoPoint& point, const std::vector<GeoPoint>& polygon) {
27  if (polygon.size() < 3) return false;
28 
29  bool inside = false;
30  int n = static_cast<int>(polygon.size());
31 
32  for (int i = 0, j = n - 1; i < n; j = i++) {
33  const auto& pi = polygon[i];
34  const auto& pj = polygon[j];
35 
36  // Check if the ray intersects with the edge
37  if (((pi.lon > point.lon) != (pj.lon > point.lon)) &&
38  (point.lat < (pj.lat - pi.lat) * (point.lon - pi.lon) / (pj.lon - pi.lon) + pi.lat)) {
39  inside = !inside;
40  }
41  }
42 
43  return inside;
44 }

49 static double distancePointToSegment(const GeoPoint& p, const GeoPoint& a, const GeoPoint& b) {
50  // Convert to approximate meters (rough approximation for small distances)
51  double lat_scale = 111000.0;
52  double lon_scale = 111000.0 * std::cos(p.lat * M_PI / 180.0);
53 
54  double px = (p.lon - a.lon) * lon_scale;
55  double py = (p.lat - a.lat) * lat_scale;
56  double bx = (b.lon - a.lon) * lon_scale;
57  double by = (b.lat - a.lat) * lat_scale;
58 
59  double t = std::max(0.0, std::min(1.0, (px * bx + py * by) / (bx * bx + by * by)));
60 
61  double closest_x = t * bx;
62  double closest_y = t * by;
63 
64  double dx = px - closest_x;
65  double dy = py - closest_y;
66 
67  return std::sqrt(dx * dx + dy * dy);
68 }

74 static double distanceToPolygonBoundary(const GeoPoint& point, const std::vector<GeoPoint>& polygon) {
75  if (polygon.size() < 3) return 0.0;
76 
77  bool inside = isPointInPolygon(point, polygon);
78 
79  double min_distance = std::numeric_limits<double>::max();
80  int n = static_cast<int>(polygon.size());
81 
82  for (int i = 0, j = n - 1; i < n; j = i++) {
83  double dist = distancePointToSegment(point, polygon[j], polygon[i]);
84  min_distance = std::min(min_distance, dist);
85  }
86 
87  // Return negative distance if inside
88  return inside ? -min_distance : min_distance;
89 }

170  violatingType = GeofenceType::KEEP_OUT;
171  }
172 
173  // Check warning distance
174  if (dist > 0 && dist < warningDistance && onApproachingCallback) {
175  onApproachingCallback(currentPosition, dist, zone.name);
176  }
177  }
178 
179  // Handle violation
180  if (anyViolation) {
181  consecutiveViolations++;
182 
183  if (consecutiveViolations >= violationThreshold && !inViolation) {
184  // Trigger violation
185  inViolation = true;
186  lastViolationFence = violatingFence;
187  totalViolations++;
188

191  violation.fenceType = violatingType;
192  violation.vehiclePosition = currentPosition;
193  violation.timestamp = std::chrono::steady_clock::now();
194  violation.description = "Vehicle violated " + violatingFence;
195 
196  lastViolation = violation;
197  hasLastViolation = true;
198  violationHistory.push_back(violation);
199 
200  // Call callbacks
201  if (onViolationCallback) {
202  onViolationCallback(violation);
203  }
204 
205  // Execute safety action
206  executeSafetyAction(violation);
207 
208  // Notify status change
209  if (onStatusChangedCallback) {
210  onStatusChangedCallback(node->getStatus());
211  }
212  }
213  } else {
214  consecutiveViolations = 0;
215  if (inViolation) {
216  inViolation = false;
217 
218  // Notify status change
219  if (onStatusChangedCallback) {
220  onStatusChangedCallback(node->getStatus());
221  }
222  }
223  }

340 void GeofenceMonitorNode::addCircularZone(
341  const GeoPoint& center,
342  float radius,
343  GeofenceType type,
344  const std::string& name) {
345 
346  Polygon zone;
347  zone.name = name.empty() ? "circular_zone" : name;
348 
349  // Create circular polygon (16 segments)
350  const int segments = 16;
351  for (int i = 0; i < segments; ++i) {
352  double angle = 2.0 * M_PI * i / segments;
353  double lat = center.lat + (radius / 111000.0) * std::cos(angle);
354  double lon = center.lon + (radius / (111000.0 * std::cos(center.lat * M_PI / 180.0))) * std::sin(angle);
355  zone.vertices.push_back({lat, lon});
356  }
357 
358  if (type == GeofenceType::KEEP_IN) {
359  addKeepInZone(zone);
360  } else {
361  addKeepOutZone(zone);
362  }
