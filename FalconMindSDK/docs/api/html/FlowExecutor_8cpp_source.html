61 bool FlowExecutor::loadFlowFromBuilder(const std::string& builder_url,
62  const std::string& project_id,
63  const std::string& flow_id) {
64 #ifdef CPPHTTPLIB_OPENSSL_SUPPORT
65  try {
66  // 解析URL
67  std::string host;
68  std::string path;
69  int port = 80;
70  bool use_https = false;
71 
72  // 简单的URL解析
73  std::string url = builder_url;
74  if (url.find("https://") == 0) {
75  use_https = true;
76  port = 443;
77  url = url.substr(8); // 移除 "https://"
78  } else if (url.find("http://") == 0) {
79  url = url.substr(7); // 移除 "http://"
80  }
81 
82  // 查找端口号
83  size_t colon_pos = url.find(':');
84  size_t slash_pos = url.find('/');
85 
86  if (colon_pos != std::string::npos && (slash_pos == std::string::npos || colon_pos < slash_pos)) {
87  host = url.substr(0, colon_pos);
88  port = std::stoi(url.substr(colon_pos + 1, slash_pos - colon_pos - 1));
89  if (slash_pos != std::string::npos) {
90  path = url.substr(slash_pos);
91  }
92  } else if (slash_pos != std::string::npos) {
93  host = url.substr(0, slash_pos);
94  path = url.substr(slash_pos);
95  } else {
96  host = url;
97  }
98 
99  // 构建请求路径
100  std::string request_path = "/projects/" + project_id + "/flows/" + flow_id + "/export";
101  if (!path.empty() && path != "/") {
102  request_path = path + request_path;
103  }
104 
105  // 创建HTTP客户端
106  httplib::Client cli(host.c_str(), port);
107  cli.set_connection_timeout(5, 0); // 5秒超时
108  cli.set_read_timeout(5, 0);
109 
110  // 发送GET请求
111  auto res = cli.Get(request_path.c_str());
112 
113  if (!res) {
114  std::cerr << "FlowExecutor: HTTP request failed (no response)" << std::endl;
115  return false;
116  }
117 
118  if (res->status != 200) {
119  std::cerr << "FlowExecutor: HTTP request failed with status: " << res->status << std::endl;
120  return false;
121  }
122 
123  // 解析响应JSON并加载Flow
124  return loadFlow(res->body);
125 
126  } catch (const std::exception& e) {
127  std::cerr << "FlowExecutor: Error loading flow from Builder: " << e.what() << std::endl;
128  return false;
129  }
130 #else
131  std::cerr << "FlowExecutor: loadFlowFromBuilder requires cpp-httplib library" << std::endl;
132  std::cerr << " Builder URL: " << builder_url << std::endl;
133  std::cerr << " Project ID: " << project_id << std::endl;
134  std::cerr << " Flow ID: " << flow_id << std::endl;
135  std::cerr << " Please enable CPPHTTPLIB_OPENSSL_SUPPORT in CMakeLists.txt" << std::endl;

140 bool FlowExecutor::parseFlowDefinition(const json& flow_json_obj) {
141  try {
142  // 使用已解析的JSON对象，避免重复解析
143  // 使用const引用减少拷贝
144  const json& j = flow_json_obj;
145 
146  // 提取基本信息
147  if (!j.contains("flow_id")) {
148  std::cerr << "FlowExecutor: Flow definition missing flow_id" << std::endl;
149  return false;
150  }
151 
152  flow_id_ = j["flow_id"].get<std::string>();
153  flow_name_ = j.value("name", "");
154  flow_version_ = j.value("version", "1.0");
155 
156  // 解析节点定义
157  if (!j.contains("nodes") || !j["nodes"].is_array()) {
158  std::cerr << "FlowExecutor: Flow definition missing or invalid nodes array" << std::endl;
159  return false;
160  }
161 
162  // 预分配空间以减少重新分配
163  node_definitions_.clear();
164  node_definitions_.reserve(j["nodes"].size());
165 
166  for (const auto& node_json : j["nodes"]) {
167  NodeDefinition node_def;
168  node_def.node_id = node_json["node_id"].get<std::string>();
169  node_def.template_id = node_json["template_id"].get<std::string>();
170 
171  // 提取parameters（如果存在）- 使用引用避免拷贝
172  if (node_json.contains("parameters")) {
173  node_def.parameters_json = node_json["parameters"]; // nlohmann::json支持移动语义
174  } else {
175  node_def.parameters_json = json::object();
176  }
177 
178  node_definitions_.push_back(std::move(node_def));
179  }
180 
181  // 解析边定义
182  if (!j.contains("edges") || !j["edges"].is_array()) {
183  std::cerr << "FlowExecutor: Flow definition missing or invalid edges array" << std::endl;
184  return false;
185  }
186 
187  // 预分配空间以减少重新分配
188  edge_definitions_.clear();
189  edge_definitions_.reserve(j["edges"].size());
190 
191  for (const auto& edge_json : j["edges"]) {
192  EdgeDefinition edge_def;
193  edge_def.edge_id = edge_json.value("edge_id", "");
194  edge_def.from_node_id = edge_json["from_node_id"].get<std::string>();
195  edge_def.from_port = edge_json["from_port"].get<std::string>();
196  edge_def.to_node_id = edge_json["to_node_id"].get<std::string>();
197  edge_def.to_port = edge_json["to_port"].get<std::string>();
198 
199  edge_definitions_.push_back(std::move(edge_def));
200  }
201 
202  std::cout << "FlowExecutor: Parsed flow definition" << std::endl;
203  std::cout << " Flow ID: " << flow_id_ << std::endl;
204  std::cout << " Flow Name: " << flow_name_ << std::endl;
205  std::cout << " Flow Version: " << flow_version_ << std::endl;
206  std::cout << " Nodes: " << node_definitions_.size() << std::endl;
207  std::cout << " Edges: " << edge_definitions_.size() << std::endl;
208 
209  return true;
210  } catch (const json::exception& e) {
211  std::cerr << "FlowExecutor: JSON parsing error: " << e.what() << std::endl;
212  return false;
213  } catch (const std::exception& e) {
214  std::cerr << "FlowExecutor: Error parsing flow definition: " << e.what() << std::endl;
215  return false;
216  }

329  if (min_alt < 0.0 || min_alt > 10000.0) {

452  try {
453  // 根据模板ID配置不同类型的节点

455  auto planner = std::dynamic_pointer_cast<mission::SearchPathPlannerNode>(node);
456  if (!planner) {
457  std::cerr << "FlowExecutor: Node is not SearchPathPlannerNode" << std::endl;
458  return false;
459  }
460 
461  // 参数格式和值范围验证
462  std::string validation_error;
463 
464  // 验证搜索区域
465  if (params_json.contains("search_area")) {
466  if (!validateSearchArea(params_json["search_area"], validation_error)) {
467  std::cerr << "FlowExecutor: Invalid search_area: " << validation_error << std::endl;
468  return false;
469  }
470  }
471 
472  // 验证搜索参数
473  if (params_json.contains("search_params")) {
474  if (!validateSearchParams(params_json["search_params"], validation_error)) {
475  std::cerr << "FlowExecutor: Invalid search_params: " << validation_error << std::endl;
476  return false;
477  }
478  }
479 
480  // 解析搜索区域（验证通过后）
481  // 使用引用避免重复访问JSON对象
482  if (params_json.contains("search_area")) {
483  mission::SearchArea area;
484  const auto& area_json = params_json["search_area"]; // 使用const引用
485 
486  // 解析多边形 - 预分配空间
487  if (area_json.contains("polygon") && area_json["polygon"].is_array()) {
488  const auto& polygon_array = area_json["polygon"];
489  area.polygon.reserve(polygon_array.size());
490  for (const auto& point_json : polygon_array) {
491  mission::GeoPoint point;
492  point.lat = point_json["lat"].get<double>();
493  point.lon = point_json["lon"].get<double>();
494  point.alt = point_json.value("alt", 0.0);
495  area.polygon.push_back(point);
496  }
497  }
498 
499  area.minAltitude = area_json.value("min_altitude", 0.0);
500  area.maxAltitude = area_json.value("max_altitude", 100.0);
501 
502  planner->setSearchArea(area);
503  }
504 
505  // 解析搜索参数（验证通过后）
506  // 使用引用避免重复访问JSON对象
507  if (params_json.contains("search_params")) {
508  mission::SearchParams params;
509  const auto& params_json_obj = params_json["search_params"]; // 使用const引用
510 
511  // 解析搜索模式 - 一次性获取并转换
512  std::string pattern_str = params_json_obj.value("pattern", "LAWN_MOWER");
513  if (pattern_str == "LAWN_MOWER") {
514  params.pattern = mission::SearchPattern::LAWN_MOWER;
515  } else if (pattern_str == "SPIRAL") {
516  params.pattern = mission::SearchPattern::SPIRAL;
517  } else if (pattern_str == "ZIGZAG") {
518  params.pattern = mission::SearchPattern::ZIGZAG;
519  } else if (pattern_str == "SECTOR") {
520  params.pattern = mission::SearchPattern::SECTOR;
521  } else if (pattern_str == "WAYPOINT_LIST") {
522  params.pattern = mission::SearchPattern::WAYPOINT_LIST;
523  } else {
524  params.pattern = mission::SearchPattern::LAWN_MOWER;
525  }
526 
527  // 一次性获取所有参数值，减少JSON访问次数
528  params.altitude = params_json_obj.value("altitude", 50.0);
529  params.speed = params_json_obj.value("speed", 10.0);
530  params.spacing = params_json_obj.value("spacing", 20.0);
531  params.loiterTime = params_json_obj.value("loiter_time", 2.0);
532  params.enableDetection = params_json_obj.value("enable_detection", false);
533 
534  planner->setSearchParams(params);
535  }
536  }
537  // 其他节点类型的参数配置可以在这里添加
538  // else if (template_id == "event_reporter") { ... }
539  // else if (template_id == "camera_source") { ... }
540 
541  return true;
542  } catch (const json::exception& e) {
543  std::cerr << "FlowExecutor: Error configuring node parameters: " << e.what() << std::endl;
544  return false;
545  } catch (const std::exception& e) {
546  std::cerr << "FlowExecutor: Error configuring node: " << e.what() << std::endl;

551 bool FlowExecutor::connectNodes() {
552  if (!pipeline_) {
553  std::cerr << "FlowExecutor: Pipeline not created" << std::endl;
554  return false;
555  }
556 
557  // 如果没有边定义，直接返回成功（允许没有连接的Flow）
558  if (edge_definitions_.empty()) {
559  return true;
560  }
561 
562  for (const auto& edge_def : edge_definitions_) {
563  auto from_it = nodes_.find(edge_def.from_node_id);
564  auto to_it = nodes_.find(edge_def.to_node_id);
565 
566  if (from_it == nodes_.end() || to_it == nodes_.end()) {
567  std::cerr << "FlowExecutor: Node not found for edge: " << edge_def.edge_id
568  << " (from: " << edge_def.from_node_id
569  << ", to: " << edge_def.to_node_id << ")" << std::endl;
570  return false;
571  }
572 
573  // 验证Pad是否存在
574  auto from_node = from_it->second;
575  auto to_node = to_it->second;
576  auto from_pad = from_node->getPad(edge_def.from_port);
577  auto to_pad = to_node->getPad(edge_def.to_port);
578 
579  if (!from_pad) {
580  std::cerr << "FlowExecutor: Source pad not found: " << edge_def.from_node_id
581  << ":" << edge_def.from_port << std::endl;
582  return false;
583  }
584 
585  if (!to_pad) {
586  std::cerr << "FlowExecutor: Sink pad not found: " << edge_def.to_node_id
