396 PoseMsg FalconMindROS2Node::toROS2Pose(
397  const GeoPoint& position, float roll, float pitch, float yaw) {
398 
399  PoseMsg pose;
400  pose.position.x = position.longitude;
401  pose.position.y = position.latitude;
402  pose.position.z = position.altitude;
403 
404  // Convert Euler to quaternion
405  float cy = std::cos(yaw * 0.5);
406  float sy = std::sin(yaw * 0.5);
407  float cp = std::cos(pitch * 0.5);
408  float sp = std::sin(pitch * 0.5);
409  float cr = std::cos(roll * 0.5);
410  float sr = std::sin(roll * 0.5);
411 
412  pose.orientation.w = cr * cp * cy + sr * sp * sy;
413  pose.orientation.x = sr * cp * cy - cr * sp * sy;
414  pose.orientation.y = cr * sp * cy + sr * cp * sy;
415  pose.orientation.z = cr * cp * sy - sr * sp * cy;
416 
417  return pose;
418 }
