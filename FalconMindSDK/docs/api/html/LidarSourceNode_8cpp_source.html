301  return false;

306  fcntl(socketFd_, F_SETFL, flags | O_NONBLOCK);
307 
308  // 允许地址重用

310  if (setsockopt(socketFd_, SOL_SOCKET, SO_REUSEADDR, &reuse, sizeof(reuse)) < 0) {

315  struct sockaddr_in addr{};

318  addr.sin_addr.s_addr = inet_addr(config_.ipAddress.c_str());

320  if (addr.sin_addr.s_addr == INADDR_NONE) {
321  addr.sin_addr.s_addr = htonl(INADDR_ANY);

323 
324  if (::bind(socketFd_, (struct sockaddr*)&addr, sizeof(addr)) < 0) {
325  std::cerr << "[LidarSourceNode] Failed to bind socket: " << strerror(errno) << std::endl;
326  close(socketFd_);

328  return false;
329  }

351 void LidarSourceNode::receiveThreadFunc() {
352 #ifdef __linux__
353  if (socketFd_ < 0) return;
354 
355  uint8_t buffer[2048];
356  struct sockaddr_in clientAddr;
357  socklen_t addrLen = sizeof(clientAddr);
358 
359  while (!stopThread_.load()) {
360  ssize_t n = ::recvfrom(socketFd_, buffer, sizeof(buffer), 0,
361  (struct sockaddr*)&clientAddr, &addrLen);
362 
363  if (n > 0) {
364  // 解析数据包
365  std::lock_guard<std::mutex> lock(packetMutex_);
366 
367  RawPacket packet;
368  packet.timestamp = std::chrono::duration_cast<std::chrono::nanoseconds>(
369  std::chrono::high_resolution_clock::now().time_since_epoch()).count();
370  packet.data.assign(buffer, buffer + n);
371  packet.sourceIp = inet_ntoa(clientAddr.sin_addr);
372  packet.sourcePort = ntohs(clientAddr.sin_port);
373 
374  packetQueue_.push(std::move(packet));
375  packetCount_++;
376 
377  // 限制队列大小
378  if (packetQueue_.size() > 100) {
379  packetQueue_.pop();
380  }
381  } else if (n < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
382  std::cerr << "[LidarSourceNode] recvfrom error: " << strerror(errno) << std::endl;
383  } else {
384  // 无数据，短暂休眠
385  std::this_thread::sleep_for(std::chrono::microseconds(100));
386  }
387  }
388 #endif
389 }

395 void LidarSourceNode::assemblyThreadFunc() {
396  PointCloud currentCloud;
397  currentCloud.reserve(10000);
398 
399  uint64_t currentFrameStart = 0;
400  const uint64_t FRAME_INTERVAL_NS = static_cast<uint64_t>(1e9 / config_.frameRateHz);
401 
402  while (!stopThread_.load()) {
403  std::unique_lock<std::mutex> lock(packetMutex_);
404 
405  if (packetQueue_.empty()) {
406  lock.unlock();
407  std::this_thread::sleep_for(std::chrono::milliseconds(1));
408  continue;
409  }
410 
411  RawPacket packet = std::move(packetQueue_.front());
412  packetQueue_.pop();
413  lock.unlock();
414 
415  // 解析数据包
416  PointCloud partialCloud;
417  bool success = false;
418 
419  switch (config_.deviceType) {
420  case LidarDeviceType::VelodyneVLP16:
421  case LidarDeviceType::VelodyneVLP32:
422  success = parseVelodynePacket(packet.data.data(), packet.data.size(), partialCloud);
423  break;
424  case LidarDeviceType::LivoxMid40:
425  case LidarDeviceType::LivoxMid70:
426  case LidarDeviceType::LivoxHorizon:
427  case LidarDeviceType::LivoxAvia:
428  success = parseLivoxPacket(packet.data.data(), packet.data.size(), partialCloud);
429  break;
430  default:
431  break;
432  }
433 
434  if (success && !partialCloud.empty()) {
435  // 检查是否需要发送当前帧
436  if (currentFrameStart == 0) {
437  currentFrameStart = packet.timestamp;
438  }
439 
440  if (packet.timestamp - currentFrameStart > FRAME_INTERVAL_NS) {
441  // 发送当前帧
442  {
443  std::lock_guard<std::mutex> qlock(queueMutex_);
444  if (cloudQueue_.size() < 10) { // 限制队列大小
445  cloudQueue_.push(std::move(currentCloud));
446  }
447  }
448 
449  currentCloud.clear();
450  currentCloud.reserve(10000);
451  currentFrameStart = packet.timestamp;
452  }
453 
454  // 添加点到当前帧
455  currentCloud.insert(currentCloud.end(), partialCloud.begin(), partialCloud.end());
456  }
457  }
458 }

464 bool LidarSourceNode::parseVelodynePacket(const uint8_t* data, size_t len, PointCloud& cloud) {
465  if (len != sizeof(VelodynePacket)) {
466  return false;
467  }
468 
469  const VelodynePacket* packet = reinterpret_cast<const VelodynePacket*>(data);
470  cloud.clear();
471  cloud.reserve(384); // 12 blocks * 32 points
472 
473  // 获取垂直角表
474  const float* verticalAngles = (config_.scanLineCount == 16) ?
475  VLP16_VERTICAL_ANGLES : VLP32_VERTICAL_ANGLES;
476  int laserCount = config_.scanLineCount;
477 
478  // 解析12个数据块
479  for (int blockIdx = 0; blockIdx < 12; ++blockIdx) {
480  const auto& block = packet->blocks[blockIdx];
481 
482  // 检查标志
483  if (block.header.flag != 0xEEFF) {
484  continue;
485  }
486 
487  // 方位角（0.01度单位）
488  float azimuth = block.header.azimuth * 0.01f * M_PI / 180.0f;
489 
490  // 解析32个激光点（每2个数据块为一组，但VLP-16只有16线）
491  for (int laserIdx = 0; laserIdx < 32; ++laserIdx) {
492  if (laserCount == 16 && laserIdx >= 16) break;
493 
494  int dataOffset = laserIdx * 3;
495  if (dataOffset + 2 >= 96) break;
496 
497  // 距离（2mm单位）
498  uint16_t distance = (block.data[dataOffset + 1] << 8) | block.data[dataOffset];
499  uint8_t intensity = block.data[dataOffset + 2];
500 
501  if (distance == 0) continue; // 无效点
502 
503  float distanceM = distance * 0.002f; // 转换为米
504  float verticalAngle = verticalAngles[laserIdx % laserCount] * M_PI / 180.0f;
505 
506  // 球坐标转笛卡尔坐标
507  PointXYZI point;
508  point.x = distanceM * std::cos(verticalAngle) * std::sin(azimuth);
509  point.y = distanceM * std::cos(verticalAngle) * std::cos(azimuth);
510  point.z = distanceM * std::sin(verticalAngle);
511  point.intensity = intensity / 255.0f;
512 
513  cloud.push_back(point);
514  }
515  }
516 
517  return !cloud.empty();
518 }

524 bool LidarSourceNode::parseLivoxPacket(const uint8_t* data, size_t len, PointCloud& cloud) {
525  if (len < sizeof(LivoxPacket)) {
526  return false;
527  }
528 
529  const LivoxPacket* packet = reinterpret_cast<const LivoxPacket*>(data);
530  cloud.clear();
531  cloud.reserve(packet->point_num);
532 
533  for (uint32_t i = 0; i < packet->point_num && i < 96; ++i) {
534  const auto& lp = packet->points[i];
535 
536  // 24-bit整数转float (mm转m)
537  int32_t x = (lp.x[0] | (lp.x[1] << 8) | (lp.x[2] << 16));
538  int32_t y = (lp.y[0] | (lp.y[1] << 8) | (lp.y[2] << 16));
539  int32_t z = (lp.z[0] | (lp.z[1] << 8) | (lp.z[2] << 16));
540 
541  // 符号扩展
542  if (x & 0x800000) x |= 0xFF000000;
543  if (y & 0x800000) y |= 0xFF000000;
544  if (z & 0x800000) z |= 0xFF000000;
545 
546  PointXYZI point;
547  point.x = x * 0.001f; // mm to m
548  point.y = y * 0.001f;
549  point.z = z * 0.001f;
550  point.intensity = lp.reflectivity / 255.0f;
551 
552  cloud.push_back(point);
553  }
554 
555  return !cloud.empty();
556 }

562 void LidarSourceNode::generateSimulatedPointCloud(PointCloud& cloud) {
563  cloud.clear();
564  cloud.reserve(config_.scanLineCount * 180); // 每线180个点
565 
566  static float time = 0.0f;
567  time += 0.1f;
568 
569  // 生成螺旋点云
570  for (int ring = 0; ring < config_.scanLineCount; ++ring) {
571  float verticalAngle = -15.0f + (30.0f * ring / (config_.scanLineCount - 1));
572  float radV = verticalAngle * M_PI / 180.0f;
573 
574  for (int i = 0; i < 180; ++i) {
575  float azimuth = i * 2.0f * M_PI / 180.0f;
576 
577  // 添加一些变化
578  float distance = 10.0f + 5.0f * std::sin(time + azimuth * 2.0f);
579 
580  PointXYZI point;
581  point.x = distance * std::cos(radV) * std::sin(azimuth);
582  point.y = distance * std::cos(radV) * std::cos(azimuth);
583  point.z = distance * std::sin(radV);
584  point.intensity = static_cast<float>(i) / 180.0f;
585 
586  cloud.push_back(point);
587  }
588  }

595 void LidarSourceNode::applyMotionCompensation(PointCloud& cloud,
596  const ImuSample& imu,
597  uint64_t targetTime) {
598  if (!config_.enableMotionCompensation || cloud.empty()) return;
599 
600  // 简化的运动补偿：基于IMU加速度估计（积分得到速度）
601  double dt = (targetTime - imu.timestampNs) * 1e-9;
602 
603  // 使用加速度积分估算位移（简化模型）
604  float dx = 0.5f * imu.ax * dt * dt;
605  float dy = 0.5f * imu.ay * dt * dt;
606  float dz = 0.5f * imu.az * dt * dt;
607 
608  for (auto& point : cloud) {
609  point.x += dx;
610  point.y += dy;
611  point.z += dz;
612  }
613 }

615 void LidarSourceNode::filterPointCloud(PointCloud& cloud, float minRange, float maxRange) {
616  if (cloud.empty()) return;
617 
618  size_t writeIdx = 0;
619  for (size_t i = 0; i < cloud.size(); ++i) {
620  const auto& p = cloud[i];
621  float range = std::sqrt(p.x * p.x + p.y * p.y + p.z * p.z);
622 
623  if (range >= minRange && range <= maxRange) {
624  cloud[writeIdx++] = p;
625  }
626  }
627 
628  cloud.resize(writeIdx);
629 }

631 void LidarSourceNode::downsamplePointCloud(PointCloud& cloud, float leafSize) {
632  if (cloud.empty() || leafSize <= 0) return;
633 
634  // 体素网格降采样（简化版）
635  struct Voxel {
636  float x = 0, y = 0, z = 0, intensity = 0;
637  int count = 0;
638  };
639 
640  std::unordered_map<uint64_t, Voxel> voxels;
641 
642  for (const auto& p : cloud) {
643  int vx = static_cast<int>(std::floor(p.x / leafSize));
644  int vy = static_cast<int>(std::floor(p.y / leafSize));
645  int vz = static_cast<int>(std::floor(p.z / leafSize));
646 
647  uint64_t key = ((static_cast<uint64_t>(vx) & 0x1FFFF) << 34) |
648  ((static_cast<uint64_t>(vy) & 0x1FFFF) << 17) |
