23 bool Pipeline::link(const std::string& srcNodeId,
24  const std::string& srcPadName,
25  const std::string& dstNodeId,
26  const std::string& dstPadName) {
27  // 检查源节点和目标节点是否存在
28  auto srcIt = nodes_.find(srcNodeId);
29  auto dstIt = nodes_.find(dstNodeId);
30  if (srcIt == nodes_.end() || dstIt == nodes_.end()) {
31  return false;
32  }
33 
34  auto srcNode = srcIt->second;
35  auto dstNode = dstIt->second;
36 
37  // 检查Pad是否存在
38  auto srcPad = srcNode->getPad(srcPadName);
39  auto dstPad = dstNode->getPad(dstPadName);
40  if (!srcPad || !dstPad) {
41  return false;
42  }
43 
44  // 检查Pad类型兼容性：Source Pad只能连接到Sink Pad
45  if (srcPad->type() != PadType::Source || dstPad->type() != PadType::Sink) {
46  return false;
47  }
48 
49  // 检查连接是否已存在
50  LinkKey key;
51  key.srcNodeId = srcNodeId;
52  key.srcPadName = srcPadName;
53  key.dstNodeId = dstNodeId;
54  key.dstPadName = dstPadName;
55 
56  if (links_.count(key) > 0) {
57  return true; // 连接已存在
58  }
59 
60  // 通过Pad的connectTo方法建立连接
61  if (!srcPad->connectTo(dstPad, dstNodeId, dstPadName)) {
62  return false;
63  }
64 
65  // 存储连接信息
66  links_.insert(key);
67 
68  return true;
69 }

79  key.dstNodeId = dstNodeId;
80  key.dstPadName = dstPadName;
81 
82  if (links_.count(key) == 0) {
83  return false; // 连接不存在
84  }
85 
86  // 获取源节点和Pad
87  auto srcIt = nodes_.find(srcNodeId);
88  if (srcIt == nodes_.end()) {
89  return false;
90  }
91 
92  auto srcPad = srcIt->second->getPad(srcPadName);
93  if (!srcPad) {
94  return false;
95  }
96 
97  // 断开Pad连接
98  srcPad->disconnect();
99 
100  // 从连接列表中移除
101  links_.erase(key);
102 
103  return true;
104 }
