12 float iou(const DetectionBBox& a, const DetectionBBox& b) {
13  float ax2 = a.x + a.width;
14  float ay2 = a.y + a.height;
15  float bx2 = b.x + b.width;
16  float by2 = b.y + b.height;
17  float ix1 = std::max(a.x, b.x);
18  float iy1 = std::max(a.y, b.y);
19  float ix2 = std::min(ax2, bx2);
20  float iy2 = std::min(ay2, by2);
21  float iw = std::max(0.f, ix2 - ix1);
22  float ih = std::max(0.f, iy2 - iy1);
23  float inter = iw * ih;
24  float areaA = a.width * a.height;
25  float areaB = b.width * b.height;
26  float u = areaA + areaB - inter;
27  return (u > 0.f) ? (inter / u) : 0.f;
28 }

76 bool SortTrackerBackend::run(DetectionResult& detections, TrackingResult& outTracks) {
77  if (!loaded_) {
78  std::cerr << "[SortTrackerBackend] run() called before load()" << std::endl;
79  return false;
80  }
81 
82  outTracks.frameId = detections.frameId;
83  outTracks.timestampNs = detections.timestampNs;
84  outTracks.frameIndex = detections.frameIndex;
85  outTracks.tracks.clear();
86 
87  const std::uint64_t ts = detections.timestampNs;
88 
89  // 1) 预测：对每条 track 用匀速模型得到预测 bbox，并增加 missed 计数
90  struct PredEntry { int id; DetectionBBox predBbox; SortTrackState* rec; };
91  std::vector<PredEntry> predictions;
92  for (auto& kv : tracks_) {
93  kv.second.missedFrames++;
94  SortTrackState& s = kv.second;
95  float px = s.cx + s.vx;
96  float py = s.cy + s.vy;
97  predictions.push_back({ kv.first, centerToBbox(px, py, s.w, s.h), &s });
98  }
99 
100  // 2) 贪心 IoU 匹配：按检测分数降序，每个检测找最大 IoU 的预测（且 > threshold）
101  std::vector<bool> detUsed(detections.detections.size(), false);
102  std::vector<bool> predUsed(predictions.size(), false);
103 
104  std::vector<size_t> detOrder(detections.detections.size());
105  for (size_t i = 0; i < detOrder.size(); ++i) detOrder[i] = i;
106  std::sort(detOrder.begin(), detOrder.end(), [&detections](size_t a, size_t b) {
107  return detections.detections[a].score > detections.detections[b].score;
108  });
109 
110  for (size_t ii = 0; ii < detOrder.size(); ++ii) {
111  size_t di = detOrder[ii];
112  if (detUsed[di]) continue;
113  Detection& det = detections.detections[di];
114  float bestIou = iouThreshold_;
115  int bestPi = -1;
116  for (size_t p = 0; p < predictions.size(); ++p) {
117  if (predUsed[p]) continue;
118  SortTrackState* rec = predictions[p].rec;
119  if (rec->missedFrames > maxMissedFrames_) continue;
120  float v = iou(predictions[p].predBbox, det.bbox);
121  if (v > bestIou) { bestIou = v; bestPi = static_cast<int>(p); }
122  }
123  if (bestPi >= 0) {
124  predUsed[static_cast<size_t>(bestPi)] = true;
125  detUsed[di] = true;
126  SortTrackState* rec = predictions[static_cast<size_t>(bestPi)].rec;
127  det.trackId = rec->trackId;
128  float nc, ny, nw, nh;
129  bboxToCenter(det.bbox, nc, ny, nw, nh);
130  rec->vx = 0.3f * rec->vx + 0.7f * (nc - rec->cx);
131  rec->vy = 0.3f * rec->vy + 0.7f * (ny - rec->cy);
132  rec->cx = nc;
133  rec->cy = ny;
134  rec->w = nw;
135  rec->h = nh;
136  rec->lastTimestampNs = ts;
137  rec->missedFrames = 0;
138  rec->classId = det.classId;
139  rec->className = det.className;
140  rec->trajectory.push_back(TrackHistoryPoint{ts, det.bbox});
141  pruneTrajectory(rec->trajectory, maxTrajectoryPoints_);
142  }
143  }
144 
145  // 3) 未匹配的检测 -> 新 track
146  for (size_t i = 0; i < detections.detections.size(); ++i) {
147  if (detUsed[i]) continue;
148  Detection& det = detections.detections[i];
149  int id = nextTrackId_++;
150  det.trackId = id;
151  SortTrackState s;
152  s.trackId = id;
153  bboxToCenter(det.bbox, s.cx, s.cy, s.w, s.h);
154  s.lastTimestampNs = ts;
155  s.missedFrames = 0;
156  s.classId = det.classId;
157  s.className = det.className;
158  s.trajectory.push_back(TrackHistoryPoint{ts, det.bbox});
159  tracks_[id] = std::move(s);
160  }
161 
162  // 4) 移除超时 track，写出 ACTIVE / LOST
163  std::vector<int> toRemove;
164  for (auto& kv : tracks_) {
165  TrackingState state;
166  state.trackId = kv.second.trackId;
167  state.targetClassId = kv.second.classId;
168  state.targetClassName = kv.second.className;
169  state.trajectory = kv.second.trajectory;
170  if (kv.second.missedFrames > maxMissedFrames_) {
171  state.status = "LOST";
172  toRemove.push_back(kv.first);
173  } else {
174  state.status = "ACTIVE";
175  }
176  outTracks.tracks.push_back(std::move(state));
177  }
178  for (int id : toRemove)
179  tracks_.erase(id);
180 
181  return true;
182 }
