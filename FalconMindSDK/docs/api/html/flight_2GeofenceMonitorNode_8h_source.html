276 inline std::shared_ptr<GeofenceMonitorNode> createCircularGeofence(
277  const std::string& id,
278  const GeoPoint& center,
279  float radius,
280  float maxAltitude = 120.0f) {
281 
282  auto node = std::make_shared<GeofenceMonitorNode>(id);
283 
284  Polygon zone;
285  zone.name = "circular_zone";
286  zone.maxAltitude = maxAltitude;
287 
288  // 创建圆形多边形（简化为16边形）
289  const int segments = 16;
290  for (int i = 0; i < segments; ++i) {
291  double angle = 2.0 * M_PI * i / segments;
292  double lat = center.lat + (radius / 111000.0) * std::cos(angle);
293  double lon = center.lon + (radius / (111000.0 * std::cos(center.lat * M_PI / 180.0))) * std::sin(angle);
294  zone.vertices.push_back({lat, lon});
295  }
296 
297  node->addKeepInZone(zone);
298  return node;
299 }
