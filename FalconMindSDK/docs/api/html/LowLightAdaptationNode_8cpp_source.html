57 // 对 RGB/BGR 三通道图做 gamma 增强：out = 255 * (in/255)^(1/gamma)
58 void applyGammaRgb(std::uint8_t* data, size_t numPixels, int channels, float gamma) {
59  if (gamma <= 0.f || gamma > 4.f) return;
60  float invGamma = 1.f / gamma;
61  for (size_t i = 0; i < numPixels * static_cast<size_t>(channels); ++i) {
62  float v = data[i] / 255.f;
63  v = std::pow(v, invGamma);
64  data[i] = static_cast<std::uint8_t>(std::min(255.f, v * 255.f + 0.5f));
65  }
66 }

68 // 采样估计亮度（均值），避免全图遍历两遍时用同一遍既算均值又做 gamma
69 float meanBrightnessRgb(const std::uint8_t* data, int width, int height, int stride, int step) {
70  double sum = 0;
71  int n = 0;
72  for (int y = 0; y < height; y += step) {
73  const std::uint8_t* row = data + static_cast<size_t>(y) * stride;
74  for (int x = 0; x < width; x += step) {
75  size_t o = static_cast<size_t>(x) * 3;
76  sum += (0.299f * row[o] + 0.587f * row[o + 1] + 0.114f * row[o + 2]);
77  n++;
78  }
79  }
80  return (n > 0) ? static_cast<float>(sum / n) : 0.f;
81 }

85 void LowLightAdaptationNode::process() {
86  if (!started_) return;
87  if (!lastFrameValid_ || lastFrameBuffer_.size() < sizeof(CameraFramePacket)) {
88  lastFrameValid_ = false;
89  return;
90  }
91 
92  auto* header = reinterpret_cast<CameraFramePacket*>(lastFrameBuffer_.data());
93  int bpp = bytesPerPixel(header->format);
94  size_t pixelBytes = (header->stride > 0 && header->height > 0)
95  ? static_cast<size_t>(header->stride) * static_cast<size_t>(header->height)
96  : static_cast<size_t>(header->width) * static_cast<size_t>(header->height) * bpp;
97  if (lastFrameBuffer_.size() < sizeof(CameraFramePacket) + pixelBytes) {
98  lastFrameValid_ = false;
99  return;
100  }
101 
102  std::uint8_t* pixels = cameraFramePacketDataWritable(header);
103  bool doEnhance = false;
104  if (isRgbOrBgr(header->format) && bpp == 3) {
105  int stride = (header->stride > 0) ? header->stride : (header->width * 3);
106  float mean = meanBrightnessRgb(pixels, header->width, header->height, stride, 4);
107  if (mean < static_cast<float>(brightnessThreshold_))
108  doEnhance = true;
109  }
110 
111  if (doEnhance)
112  applyGammaRgb(pixels, static_cast<size_t>(header->width) * static_cast<size_t>(header->height), bpp, gamma_);
113 
114  auto outPad = getPad("image_out");
115  if (outPad)
116  outPad->pushToConnections(lastFrameBuffer_.data(), lastFrameBuffer_.size());
117  lastFrameValid_ = false;
118 }
