29 bool RknnDetectorBackend::load(const DetectorDescriptor& desc) {
30  desc_ = desc;
31 #if defined(FALCONMINDSDK_RKNN_BACKEND_ENABLED) && FALCONMINDSDK_RKNN_BACKEND_ENABLED
32  if (rknnState_) {
33  unload();
34  }
35  auto* state = new RknnState();
36  state->inputW = desc_.inputWidth > 0 ? desc_.inputWidth : 640;
37  state->inputH = desc_.inputHeight > 0 ? desc_.inputHeight : 640;
38  state->inputSize = 1 * 3 * static_cast<uint32_t>(state->inputH) * static_cast<uint32_t>(state->inputW) * sizeof(float);
39 
40  // rknn_init: size=0 表示 model 为文件路径
41  int ret = rknn_init(&state->ctx, (void*)desc_.modelPath.c_str(), 0, 0, nullptr);
42  if (ret != RKNN_SUCC) {
43  std::cerr << "[RknnDetectorBackend] rknn_init failed: " << ret << " path=" << desc_.modelPath << std::endl;
44  delete state;
45  return false;
46  }
47 
48  rknn_input_output_num num;
49  ret = rknn_query(state->ctx, RKNN_QUERY_IN_OUT_NUM, &num, sizeof(num));
50  if (ret != RKNN_SUCC || num.n_input < 1) {
51  std::cerr << "[RknnDetectorBackend] rknn_query IN_OUT_NUM failed or no input" << std::endl;
52  rknn_destroy(state->ctx);
53  delete state;
54  return false;
55  }
56 
57  rknn_tensor_attr attr;
58  memset(&attr, 0, sizeof(attr));
59  attr.index = 0;
60  ret = rknn_query(state->ctx, RKNN_QUERY_INPUT_ATTR, &attr, sizeof(attr));
61  if (ret == RKNN_SUCC && attr.n_dims >= 2) {
62  // 常见 NCHW: dims = [1,3,H,W] 或 NHWC: [1,H,W,3]
63  if (attr.fmt == RKNN_TENSOR_NCHW && attr.n_dims >= 4) {
64  state->inputH = static_cast<int>(attr.dims[2]);
65  state->inputW = static_cast<int>(attr.dims[3]);
66  } else if (attr.fmt == RKNN_TENSOR_NHWC && attr.n_dims >= 4) {
67  state->inputH = static_cast<int>(attr.dims[1]);
68  state->inputW = static_cast<int>(attr.dims[2]);
69  }
70  state->inputFmt = attr.fmt;
71  state->inputType = attr.type;
72  state->inputSize = attr.size;
73  }
74  if (state->inputSize == 0)
75  state->inputSize = 1 * 3 * static_cast<uint32_t>(state->inputH) * static_cast<uint32_t>(state->inputW) * sizeof(float);
76 
77  rknnState_ = state;
78  loaded_ = true;
79  std::cout << "[RknnDetectorBackend] loaded: " << desc_.modelPath
80  << " " << state->inputW << "x" << state->inputH
81  << " fmt=" << (state->inputFmt == RKNN_TENSOR_NCHW ? "NCHW" : "NHWC") << std::endl;
82  return true;
83 #else
84  loaded_ = true;
85  std::cout << "[RknnDetectorBackend] load (stub): " << desc_.modelPath
86  << " (id=" << desc_.detectorId << ")" << std::endl;
87  return true;
88 #endif
89 }

201  float scaleY = (image.height > 0 && inputH > 0) ? (image.height / static_cast<float>(inputH)) : 1.0f;
202  fillDetectionResultFromYolo(raw, suppressed, scaleX, scaleY, outResult);
203  return true;
204 #else
