91 bool SearchMissionAction::isWaypointReached(const GeoPoint& waypoint,
92  const GeoPoint& currentPos,
93  double tolerance) {
94  // Haversine距离计算
95  const double lat1 = waypoint.lat * M_PI / 180.0;
96  const double lat2 = currentPos.lat * M_PI / 180.0;
97  const double dLat = (currentPos.lat - waypoint.lat) * M_PI / 180.0;
98  const double dLon = (currentPos.lon - waypoint.lon) * M_PI / 180.0;
99 
100  const double a = std::sin(dLat / 2) * std::sin(dLat / 2) +
101  std::cos(lat1) * std::cos(lat2) *
102  std::sin(dLon / 2) * std::sin(dLon / 2);
103  const double c = 2 * std::atan2(std::sqrt(a), std::sqrt(1 - a));
104  const double distance = 6371000.0 * c; // 地球半径（米）
105 
106  const double altDiff = std::abs(currentPos.alt - waypoint.alt);
107 
108  return distance < tolerance && altDiff < tolerance;
109 }

111 bool SearchMissionAction::uploadMissionToPX4(const std::vector<GeoPoint>& waypoints) {
112  if (waypoints.empty()) {
113  std::cerr << "[SearchMissionAction] No waypoints to upload" << std::endl;
114  return false;
115  }
116 
117  std::cout << "[SearchMissionAction] Uploading " << waypoints.size()
118  << " waypoints to PX4..." << std::endl;
119 
120  // 1. 发送航点计数
121  if (!sendMissionCount(waypoints.size())) {
122  std::cerr << "[SearchMissionAction] Failed to send mission count" << std::endl;
123  return false;
124  }
125 
126  // 2. 等待并响应航点请求，逐个发送航点
127  for (size_t i = 0; i < waypoints.size(); ++i) {
128  if (!waitAndSendWaypoint(static_cast<int>(i), waypoints[i])) {
129  std::cerr << "[SearchMissionAction] Failed to send waypoint " << i << std::endl;
130  return false;
131  }
132 
133  // 上报进度
134  SearchProgress progress;
135  progress.coveragePercent = static_cast<double>(i) / waypoints.size();
136  progress.waypointIndex = static_cast<int>(i);
137  progress.totalWaypoints = static_cast<int>(waypoints.size());
138  progress.currentPosition = {waypoints[i].lat, waypoints[i].lon, waypoints[i].alt};
139 
140  if (eventReporter_) {
141  eventReporter_->reportSearchProgress(progress);
142  }
143  }
144 
145  // 3. 等待任务确认
146  if (!waitForMissionAck()) {
147  std::cerr << "[SearchMissionAction] Mission upload not acknowledged" << std::endl;
148  return false;
149  }
150 
151  std::cout << "[SearchMissionAction] Mission upload successful" << std::endl;
152  return true;

155 bool SearchMissionAction::sendMissionCount(int count) {
156  // 构建MAVLink MISSION_COUNT消息
157  uint8_t buffer[32];
158  uint16_t len = 0;
159 
160  // 消息头
161  buffer[len++] = 0xFE; // MAVLink v1起始字节
162  buffer[len++] = 4; // Payload长度
163  buffer[len++] = 0; // Packet sequence
164  buffer[len++] = 1; // System ID
165  buffer[len++] = 1; // Component ID
166  buffer[len++] = mavlink::MAVLINK_MSG_ID_MISSION_COUNT & 0xFF;
167  buffer[len++] = (mavlink::MAVLINK_MSG_ID_MISSION_COUNT >> 8) & 0xFF;
168 
169  // Payload: count (2 bytes), target_system, target_component
170  buffer[len++] = count & 0xFF;
171  buffer[len++] = (count >> 8) & 0xFF;
172  buffer[len++] = 1; // target_system
173  buffer[len++] = 1; // target_component
174  buffer[len++] = 0; // mission_type (0=MAV_MISSION_TYPE_MISSION)
175 
176  // 计算校验和（简化版，实际应使用MAVLink库）
177  uint8_t checksum = 0;
178  for (uint16_t i = 1; i < len; ++i) {
179  checksum ^= buffer[i];
180  }
181  buffer[len++] = checksum;
182 
183  // 发送
184  FlightCommand cmd;
185  cmd.type = FlightCommandType::SendMavlinkRaw;
186  cmd.mavlinkData.assign(buffer, buffer + len);
187  flightSvc_.sendCommand(cmd);
188 
189  std::cout << "[SearchMissionAction] Sent MISSION_COUNT: " << count << std::endl;
190  return true;
191 }

193 bool SearchMissionAction::waitAndSendWaypoint(int seq, const GeoPoint& wp) {
194  // 简化实现：直接发送航点
195  // 实际应等待MISSION_REQUEST，然后发送对应的航点
196 
197  uint8_t buffer[64];
198  uint16_t len = 0;
199 
200  // 消息头
201  buffer[len++] = 0xFE;
202  buffer[len++] = 38; // MISSION_ITEM_INT payload长度
203  buffer[len++] = static_cast<uint8_t>(seq);
204  buffer[len++] = 1; // System ID
205  buffer[len++] = 1; // Component ID
206  buffer[len++] = mavlink::MAVLINK_MSG_ID_MISSION_ITEM_INT & 0xFF;
207  buffer[len++] = (mavlink::MAVLINK_MSG_ID_MISSION_ITEM_INT >> 8) & 0xFF;
208 
209  // Payload
210  buffer[len++] = 1; // target_system
211  buffer[len++] = 1; // target_component
212  buffer[len++] = seq & 0xFF; // seq
213  buffer[len++] = (seq >> 8) & 0xFF;
214  buffer[len++] = mavlink::MAV_FRAME_GLOBAL_RELATIVE_ALT_INT; // frame
215  buffer[len++] = mavlink::MAV_CMD_NAV_WAYPOINT & 0xFF; // command
216  buffer[len++] = (mavlink::MAV_CMD_NAV_WAYPOINT >> 8) & 0xFF;
217  buffer[len++] = 1; // current (0=not current, 1=current)
218  buffer[len++] = 1; // autocontinue
219 
220  // param1-4 (float32)
221  float holdTime = searchParams_.loiterTime;
222  memcpy(&buffer[len], &holdTime, 4); len += 4; // hold time
223  float acceptRadius = 5.0f;
224  memcpy(&buffer[len], &acceptRadius, 4); len += 4; // accept radius
225  float passRadius = 0.0f;
226  memcpy(&buffer[len], &passRadius, 4); len += 4;
227  float yaw = NAN;
228  memcpy(&buffer[len], &yaw, 4); len += 4;
229 
230  // x, y, z (int32 lat/lon in degE7, float alt)
231  int32_t latE7 = static_cast<int32_t>(wp.lat * 1e7);
232  int32_t lonE7 = static_cast<int32_t>(wp.lon * 1e7);
233  memcpy(&buffer[len], &latE7, 4); len += 4;
234  memcpy(&buffer[len], &lonE7, 4); len += 4;
235  memcpy(&buffer[len], &wp.alt, 4); len += 4;
236 
237  buffer[len++] = 0; // mission_type
238 
239  // 校验和
240  uint8_t checksum = 0;
241  for (uint16_t i = 1; i < len; ++i) {
242  checksum ^= buffer[i];
243  }
244  buffer[len++] = checksum;
245 
246  // 发送
247  FlightCommand cmd;
248  cmd.type = FlightCommandType::SendMavlinkRaw;
249  cmd.mavlinkData.assign(buffer, buffer + len);
250  flightSvc_.sendCommand(cmd);
251 
252  std::cout << "[SearchMissionAction] Sent waypoint " << seq << ": ("
253  << wp.lat << ", " << wp.lon << ", " << wp.alt << ")" << std::endl;
254 
255  return true;
256 }

275 NodeStatus SearchMissionAction::executeWaypointMission() {
276  // 获取当前飞行状态
277  FlightState currentState = flightSvc_.getLastState();
278 
279  // 获取航点列表
280  const auto& waypoints = pathPlanner_->getWaypoints();
281  if (waypoints.empty()) {
282  std::cerr << "[SearchMissionAction] No waypoints available" << std::endl;
283  return NodeStatus::Failure;
284  }
285 
286  if (currentWaypointIndex_ >= static_cast<int>(waypoints.size())) {
287  std::cout << "[SearchMissionAction] All waypoints completed" << std::endl;
288  return NodeStatus::Success;
289  }
290 
291  const auto& targetWaypoint = waypoints[currentWaypointIndex_];
292 
293  // 检查是否到达航点
294  GeoPoint currentPos{currentState.lat, currentState.lon, currentState.alt};
295  double tolerance = searchParams_.waypointTolerance > 0 ? searchParams_.waypointTolerance : 5.0;
296 
297  if (isWaypointReached(targetWaypoint, currentPos, tolerance)) {
298  // 到达航点
299  auto now = std::chrono::system_clock::now();
300  auto duration = now.time_since_epoch();
301  uint64_t timestampNs = std::chrono::duration_cast<std::chrono::nanoseconds>(duration).count();
302 
303  // 上报事件
304  SearchEvent event;
305  event.type = SearchEventType::WAYPOINT_REACHED;
306  event.description = "Reached waypoint " + std::to_string(currentWaypointIndex_) +
307  "/" + std::to_string(waypoints.size());
308  event.position = targetWaypoint;
309  event.timestampNs = timestampNs;
310 
311  if (eventReporter_) {
312  eventReporter_->reportSearchEvent(event);
313  }
314 
315  std::cout << "[SearchMissionAction] ✓ Waypoint " << currentWaypointIndex_
316  << " reached" << std::endl;
317 
318  // 切换到下一个航点
319  currentWaypointIndex_++;
320  waypointRetryCount_ = 0;
321  lastWaypointTime_ = timestampNs;
322 
323  // 更新进度
324  SearchProgress progress;
325  progress.coveragePercent = static_cast<double>(currentWaypointIndex_) / waypoints.size();
326  progress.waypointIndex = currentWaypointIndex_;
327  progress.totalWaypoints = static_cast<int>(waypoints.size());
328  progress.currentPosition = currentPos;
329 
330  if (eventReporter_) {
331  eventReporter_->reportSearchProgress(progress);
332  }
333 
334  // 检查是否完成
335  if (currentWaypointIndex_ >= static_cast<int>(waypoints.size())) {
336  return NodeStatus::Success;
337  }
338  } else {
339  // 检查航点超时
340  auto now = std::chrono::system_clock::now();
341  auto duration = now.time_since_epoch();
342  uint64_t currentTimeNs = std::chrono::duration_cast<std::chrono::nanoseconds>(duration).count();
343 
344  if (lastWaypointTime_ > 0 &&
345  (currentTimeNs - lastWaypointTime_) > 60000000000ULL) { // 60秒超时
346  waypointRetryCount_++;
347 
348  if (waypointRetryCount_ > maxWaypointRetries_) {
349  std::cerr << "[SearchMissionAction] Waypoint " << currentWaypointIndex_
350  << " timeout after " << maxWaypointRetries_ << " retries" << std::endl;
351 
352  SearchEvent event;
353  event.type = SearchEventType::WAYPOINT_TIMEOUT;
354  event.description = "Waypoint timeout: " + std::to_string(currentWaypointIndex_);
355  event.position = targetWaypoint;
356  event.timestampNs = currentTimeNs;
357 
358  if (eventReporter_) {
359  eventReporter_->reportSearchEvent(event);
360  }
361 
362  // 跳过这个航点，继续下一个
363  currentWaypointIndex_++;
364  waypointRetryCount_ = 0;
365  lastWaypointTime_ = currentTimeNs;
366  } else {
367  // 重试：重新发送当前航点
368  std::cout << "[SearchMissionAction] Retrying waypoint " << currentWaypointIndex_
369  << " (attempt " << waypointRetryCount_ << "/" << maxWaypointRetries_ << ")" << std::endl;
370  lastWaypointTime_ = currentTimeNs;
371  }
372  }
373  }
374 
375  return NodeStatus::Running;
376 }

378 NodeStatus SearchMissionAction::tick() {
379  switch (state_) {
380  case MissionState::IDLE: {
381  // 配置路径规划器
382  if (pathPlanner_) {
383  pathPlanner_->setSearchArea(searchArea_);
384  pathPlanner_->setSearchParams(searchParams_);
385  // Generate waypoints based on search pattern
386  switch (searchParams_.pattern) {
387  case SearchPattern::LAWN_MOWER:
388  pathPlanner_->generateLawnMowerPath();
389  break;
390  case SearchPattern::SPIRAL:
391  pathPlanner_->generateSpiralPath();
392  break;
393  case SearchPattern::ZIGZAG:
394  pathPlanner_->generateZigzagPath();
395  break;
396  case SearchPattern::SECTOR:
397  pathPlanner_->generateSectorPath();
398  break;
399  default:
400  pathPlanner_->generateLawnMowerPath();
401  break;
402  }
403  }
404 
405  missionStartTime_ = std::chrono::duration_cast<std::chrono::nanoseconds>(
406  std::chrono::system_clock::now().time_since_epoch()).count();
407 
408  std::cout << "[SearchMissionAction] Starting search mission..." << std::endl;
409  state_ = MissionState::ARMING;
410  return NodeStatus::Running;
411  }
412 
413  case MissionState::ARMING: {
414  if (!armingDone_) {
415  FlightCommand cmd;
416  cmd.type = FlightCommandType::Arm;
417  flightSvc_.sendCommand(cmd);
418  armingDone_ = true;
419 
420  std::cout << "[SearchMissionAction] Arming..." << std::endl;
421  }
422 
423  // 检查ARM状态（简化：等待1秒）
424  std::this_thread::sleep_for(std::chrono::milliseconds(500));
425 
426  FlightState state = flightSvc_.getLastState();
427  if (state.armed) {
428  std::cout << "[SearchMissionAction] Armed successfully" << std::endl;
429  state_ = MissionState::TAKING_OFF;
430  }
431  return NodeStatus::Running;
432  }
433 
434  case MissionState::TAKING_OFF: {
435  if (!takeoffDone_) {
436  FlightCommand cmd;
437  cmd.type = FlightCommandType::Takeoff;
438  cmd.targetAlt = searchParams_.altitude;
439  flightSvc_.sendCommand(cmd);
440  takeoffDone_ = true;
441 
442  std::cout << "[SearchMissionAction] Taking off to " << searchParams_.altitude << "m" << std::endl;
443  }
444 
445  // 检查起飞完成
446  FlightState state = flightSvc_.getLastState();
447  if (std::abs(state.alt - searchParams_.altitude) < 2.0) {
448  std::cout << "[SearchMissionAction] Takeoff complete" << std::endl;
449  state_ = MissionState::UPLOADING_MISSION;
450  }
451  return NodeStatus::Running;
452  }
453 
454  case MissionState::UPLOADING_MISSION: {
455  if (!waypointsUploaded_) {
456  const auto& waypoints = pathPlanner_->getWaypoints();
457  if (waypoints.empty()) {
458  std::cerr << "[SearchMissionAction] No waypoints generated" << std::endl;
459  return NodeStatus::Failure;
460  }
461 
462  if (uploadMissionToPX4(waypoints)) {
463  waypointsUploaded_ = true;
464  state_ = MissionState::EXECUTING_MISSION;
465 
466  // 开始执行
467  startMissionExecution();
468  } else {
469  return NodeStatus::Failure;
470  }
471  }
472  return NodeStatus::Running;
473  }
474 
475  case MissionState::EXECUTING_MISSION:
476  case MissionState::SEARCHING: {
477  NodeStatus status = executeWaypointMission();
478 
479  if (status == NodeStatus::Success) {
480  // 搜索完成
481  auto now = std::chrono::system_clock::now();
482  auto duration = now.time_since_epoch();
483  uint64_t timestampNs = std::chrono::duration_cast<std::chrono::nanoseconds>(duration).count();
484 
485  SearchEvent event;
486  event.type = SearchEventType::SEARCH_COMPLETE;
487  event.description = "Search mission completed successfully";
488  event.position = {0, 0, 0}; // 最终位置
489  event.timestampNs = timestampNs;
490 
491  if (eventReporter_) {
492  eventReporter_->reportSearchEvent(event);
493  }
494 
495  std::cout << "[SearchMissionAction] Search mission completed!" << std::endl;
496  state_ = MissionState::RETURNING;
497  }
498  return status;
499  }
500 
501  case MissionState::RETURNING: {
502  FlightCommand cmd;
503  cmd.type = FlightCommandType::ReturnToLaunch;
504  flightSvc_.sendCommand(cmd);
505 
506  std::cout << "[SearchMissionAction] Returning to launch" << std::endl;
507  state_ = MissionState::LANDING;
508  return NodeStatus::Running;
509  }
510 
511  case MissionState::LANDING: {
512  // 检查是否降落完成
513  FlightState state = flightSvc_.getLastState();
514  if (state.alt < 1.0 && std::abs(state.vz) < 0.5) {
515  std::cout << "[SearchMissionAction] Landed" << std::endl;
516  state_ = MissionState::DISARMING;
517  }
518  return NodeStatus::Running;
519  }
520 
521  case MissionState::DISARMING: {
522  FlightCommand cmd;
523  cmd.type = FlightCommandType::Disarm;
524  flightSvc_.sendCommand(cmd);
525 
526  std::cout << "[SearchMissionAction] Disarming" << std::endl;
527  state_ = MissionState::COMPLETE;
528  return NodeStatus::Running;
529  }
530 
531  case MissionState::COMPLETE: {
532  // 计算总耗时
533  auto now = std::chrono::system_clock::now();
534  auto duration = now.time_since_epoch();
535  uint64_t elapsedMs = (std::chrono::duration_cast<std::chrono::nanoseconds>(duration).count() - missionStartTime_) / 1000000;
536 
537  std::cout << "[SearchMissionAction] Mission complete in " << elapsedMs / 1000.0 << "s" << std::endl;
538  return NodeStatus::Success;
539  }
540  }
541 
542  return NodeStatus::Failure;
