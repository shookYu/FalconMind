26 // 计算点曲率（用于特征提取）
27 float computeCurvature(const std::vector<Point3D>& points, int idx, int windowSize = 5) {
28  int start = std::max(0, idx - windowSize);
29  int end = std::min(static_cast<int>(points.size()) - 1, idx + windowSize);
30 
31  if (end - start < 2) return 0.0f;
32 
33  const auto& p = points[idx];
34  float sumDistance = 0.0f;
35 
36  for (int i = start; i <= end; ++i) {
37  if (i != idx) {
38  const auto& neighbor = points[i];
39  float dx = p.x - neighbor.x;
40  float dy = p.y - neighbor.y;
41  float dz = p.z - neighbor.z;
42  sumDistance += std::sqrt(dx*dx + dy*dy + dz*dz);
43  }
44  }
45 
46  return sumDistance / (end - start);
47 }

50 float pointToLineDistance(const Point3D& point,
51  const Point3D& lineStart,
52  const Point3D& lineEnd) {
53  float ux = lineEnd.x - lineStart.x;
54  float uy = lineEnd.y - lineStart.y;
55  float uz = lineEnd.z - lineStart.z;
56 
57  float vx = point.x - lineStart.x;
58  float vy = point.y - lineStart.y;
59  float vz = point.z - lineStart.z;
60 
61  // 叉积模
62  float cx = uy * vz - uz * vy;
63  float cy = uz * vx - ux * vz;
64  float cz = ux * vy - uy * vx;
65 
66  float crossNorm = std::sqrt(cx*cx + cy*cy + cz*cz);
67  float lineNorm = std::sqrt(ux*ux + uy*uy + uz*uz);
68 
69  if (lineNorm < 1e-6f) return 0.0f;
70  return crossNorm / lineNorm;
71 }

74 float pointToPlaneDistance(const Point3D& point,
75  const Point3D& p1,
76  const Point3D& p2,
77  const Point3D& p3) {
78  // 计算平面法向量
79  float ux = p2.x - p1.x, uy = p2.y - p1.y, uz = p2.z - p1.z;
80  float vx = p3.x - p1.x, vy = p3.y - p1.y, vz = p3.z - p1.z;
81 
82  float nx = uy * vz - uz * vy;
83  float ny = uz * vx - ux * vz;
84  float nz = ux * vy - uy * vx;
85  float norm = std::sqrt(nx*nx + ny*ny + nz*nz);
86 
87  if (norm < 1e-6f) return 0.0f;
88 
89  nx /= norm; ny /= norm; nz /= norm;
90 
91  float dx = point.x - p1.x;
92  float dy = point.y - p1.y;
93  float dz = point.z - p1.z;
94 
95  return std::abs(dx * nx + dy * ny + dz * nz);
96 }

107 void rotatePoint(const double q[4], const Point3D& in, Point3D& out) {
108  double x = in.x, y = in.y, z = in.z;
109  double qw = q[0], qx = q[1], qy = q[2], qz = q[3];
110 
111  // q * p * q^-1
112  double tx = 2.0 * (qy * z - qz * y);
113  double ty = 2.0 * (qz * x - qx * z);
114  double tz = 2.0 * (qx * y - qy * x);
115 
116  out.x = x + qw * tx + qy * tz - qz * ty;
117  out.y = y + qw * ty + qz * tx - qx * tz;
118  out.z = z + qw * tz + qx * ty - qy * tx;
119 }

272 void LidarSlamNode::extractFeatures(const PointCloud& cloud,
273  PointCloud& edgeFeatures,
274  PointCloud& planarFeatures) {
275  edgeFeatures.clear();
276  planarFeatures.clear();
277 
278  // 按扫描线分组
279  std::vector<std::vector<size_t>> scanLines(scanLineCount_);
280  for (size_t i = 0; i < cloud.size(); ++i) {
281  int ring = cloud[i].ring % scanLineCount_;
282  scanLines[ring].push_back(i);
283  }
284 
285  // 提取特征
286  for (int ring = 0; ring < scanLineCount_; ++ring) {
287  const auto& indices = scanLines[ring];
288  if (indices.size() < 10) continue;
289 
290  // 计算每个点的曲率
291  std::vector<std::pair<float, size_t>> curvatures;
292  curvatures.reserve(indices.size());
293 
294  for (size_t i = 5; i < indices.size() - 5; ++i) {
295  float curvature = computeCurvature(cloud, indices[i]);
296  curvatures.emplace_back(curvature, indices[i]);
297  }
298 
299  // 按曲率排序
300  std::sort(curvatures.begin(), curvatures.end());
301 
302  // 选择边缘特征（高曲率）和平面特征（低曲率）
303  int edgeCount = 0, planarCount = 0;
304  int maxFeaturesPerLine = 20;
305 
306  for (const auto& [curvature, idx] : curvatures) {
307  if (curvature > edgeThreshold_ && edgeCount < maxFeaturesPerLine) {
308  edgeFeatures.push_back(cloud[idx]);
309  edgeCount++;
310  } else if (curvature < planarThreshold_ && planarCount < maxFeaturesPerLine) {
311  planarFeatures.push_back(cloud[idx]);
312  planarCount++;
313  }
314 
315  if (edgeCount >= maxFeaturesPerLine && planarCount >= maxFeaturesPerLine)
316  break;
317  }
318  }

321 bool LidarSlamNode::estimatePose(const PointCloud& edgeFeatures,
322  const PointCloud& planarFeatures,
323  Pose3D& pose) {
324  if (localMap_.empty()) {
325  // 第一帧，直接返回初始位姿
326  return true;
327  }
328 
329  // 简化版位姿估计：基于最近邻匹配的ICP
330  // 实际实现应使用Levenberg-Marquardt优化
331 
332  double bestTx = 0.0, bestTy = 0.0, bestTz = 0.0;
333  float bestScore = std::numeric_limits<float>::max();
334 
335  // 网格搜索（简化版）
336  for (double tx = -0.5; tx <= 0.5; tx += 0.1) {
337  for (double ty = -0.5; ty <= 0.5; ty += 0.1) {
338  float score = 0.0f;
339 
340  // 计算匹配分数
341  for (const auto& feat : planarFeatures) {
342  // 变换特征点
343  Point3D transformed;
344  transformed.x = feat.x + tx;
345  transformed.y = feat.y + ty;
346  transformed.z = feat.z;
347 
348  // 找最近邻
349  float minDist = std::numeric_limits<float>::max();
350  for (const auto& mapPoint : localMap_) {
351  float dx = transformed.x - mapPoint.x;
352  float dy = transformed.y - mapPoint.y;
353  float dz = transformed.z - mapPoint.z;
354  float dist = std::sqrt(dx*dx + dy*dy + dz*dz);
355  if (dist < minDist) minDist = dist;
356  }
357 
358  score += minDist;
359  }
360 
361  if (score < bestScore) {
362  bestScore = score;
363  bestTx = tx;
364  bestTy = ty;
365  }
366  }
367  }
368 
369  // 更新位姿
370  pose.x += bestTx;
371  pose.y += bestTy;
372  pose.timestampNs = std::chrono::duration_cast<std::chrono::nanoseconds>(
373  std::chrono::high_resolution_clock::now().time_since_epoch()).count();
374 
375  return true;
376 }
