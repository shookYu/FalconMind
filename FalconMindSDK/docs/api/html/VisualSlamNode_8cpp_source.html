114 void VisualSlamNode::onImageData(const void* data, size_t size) {
115  if (!vinsAdapter_ || !started_) return;
116 
117  // 解析CameraFramePacket
118  if (size < sizeof(CameraFramePacket)) return;
119 
120  const auto* packet = static_cast<const CameraFramePacket*>(data);
121  const uint8_t* imageData = static_cast<const uint8_t*>(data) + sizeof(CameraFramePacket);
122 
123  double timestamp = packet->.timestampNs / 1e9;
124 
125  // 输入到VINS-Fusion
126  vinsAdapter_->inputImage(timestamp, imageData, nullptr);
127 }
