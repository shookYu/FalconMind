22 bool GeofenceZone::contains(const GeoPoint& point) const {
23  if (polygon.size() < 3) return false;
24 
25  // Ray casting algorithm
26  bool inside = false;
27  for (size_t i = 0, j = polygon.size() - 1; i < polygon.size(); j = i++) {
28  const GeoPoint& vi = polygon[i];
29  const GeoPoint& vj = polygon[j];
30 
31  // Check if the ray intersects with this edge
32  if (((vi.lon > point.lon) != (vj.lon > point.lon)) &&
33  (point.lat < (vj.lat - vi.lat) * (point.lon - vi.lon) / (vj.lon - vi.lon) + vi.lat)) {
34  inside = !inside;
35  }
36  }
37  return inside;
38 }

232 double GeofenceMonitorNode::calculateDistanceToZone(const GeofenceZone& zone,
233  const GeoPoint& position) const {
234  if (zone.contains(position)) {
235  return 0.0;
236  }
237 
238  // Simplified distance calculation - distance to nearest vertex
239  // In production, this should calculate distance to nearest edge
240  double minDist = std::numeric_limits<double>::max();
241 
242  for (const auto& vertex : zone.polygon) {
243  // Simple Euclidean distance in lat/lon space (not accurate for large distances)
244  double dlat = vertex.lat - position.lat;
245  double dlon = vertex.lon - position.lon;
246  double dist = std::sqrt(dlat * dlat + dlon * dlon);
247 
248  // Rough conversion to meters (at equator, 1 degree ≈ 111km)
249  dist *= 111000.0;
250 
251  minDist = std::min(minDist, dist);
252  }
253 
254  return minDist;
255 }
