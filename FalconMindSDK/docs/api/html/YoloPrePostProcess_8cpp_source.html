14 float boxIou(float x1, float y1, float w1, float h1, float x2, float y2, float w2, float h2) {
15  float xa = std::max(x1, x2);
16  float ya = std::max(y1, y2);
17  float xb = std::min(x1 + w1, x2 + w2);
18  float yb = std::min(y1 + h1, y2 + h2);
19  float inter = std::max(0.0f, xb - xa) * std::max(0.0f, yb - ya);
20  float a1 = w1 * h1;
21  float a2 = w2 * h2;
22  return (a1 > 0 && a2 > 0) ? (inter / (a1 + a2 - inter)) : 0.0f;
23 }

35  int iy0 = static_cast<int>(std::floor(sy));
36  int iy1 = std::min(iy0 + 1, srcH - 1);
37  float fy = sy - iy0;
38  for (int dx = 0; dx < dstW; ++dx) {
39  float sx = (srcW > 1) ? (dx * (srcW - 1) / static_cast<float>(dstW - 1)) : 0;
40  int ix0 = static_cast<int>(std::floor(sx));
41  int ix1 = std::min(ix0 + 1, srcW - 1);
42  float fx = sx - ix0;
43  for (int c = 0; c < C; ++c) {
44  int ch = bgr ? (2 - c) : c;
45  float v00 = src[iy0 * srcStride + ix0 * C + ch];
46  float v10 = src[iy0 * srcStride + ix1 * C + ch];
47  float v01 = src[iy1 * srcStride + ix0 * C + ch];
48  float v11 = src[iy1 * srcStride + ix1 * C + ch];
49  float v = (1 - fx) * (1 - fy) * v00 + fx * (1 - fy) * v10
50  + (1 - fx) * fy * v01 + fx * fy * v11;
51  dst[c * (dstH * dstW) + dy * dstW + dx] = v / 255.0f;
