17 struct AppearanceFeature {
18  float data[128];
19 
20  float cosineDistance(const AppearanceFeature& other) const {
21  float dot = 0.0f, normA = 0.0f, normB = 0.0f;
22  for (int i = 0; i < 128; ++i) {
23  dot += data[i] * other.data[i];
24  normA += data[i] * data[i];
25  normB += other.data[i] * other.data[i];
26  }
27  return 1.0f - dot / (std::sqrt(normA) * std::sqrt(normB) + 1e-6f);
28  }
29 };
