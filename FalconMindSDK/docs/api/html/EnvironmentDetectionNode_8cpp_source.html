27 class SlidingWindow {
28 public:
29  explicit SlidingWindow(size_t size) : maxSize_(size) {}
30 
31  void add(float value) {
32  values_.push_back(value);
33  if (values_.size() > maxSize_) {
34  values_.erase(values_.begin());
35  }
36  }
37 
38  float mean() const {
39  if (values_.empty()) return 0.0f;
40  return std::accumulate(values_.begin(), values_.end(), 0.0f) / values_.size();
41  }
42 
43  float variance() const {
44  if (values_.size() < 2) return 0.0f;
45  float m = mean();
46  float sum = 0.0f;
47  for (float v : values_) {
48  sum += (v - m) * (v - m);
49  }
50  return sum / values_.size();
51  }
52 
53  float stdDev() const {
54  return std::sqrt(variance());
55  }
56 
57  void clear() { values_.clear(); }
58  size_t size() const { return values_.size(); }
59 
60 private:
61  size_t maxSize_;
62  std::vector<float> values_;
63 };

130 void EnvironmentDetectionNode::onImageData(const void* data, size_t size) {
131  if (size < sizeof(CameraFramePacket)) return;
132 
133  const auto* packet = static_cast<const CameraFramePacket*>(data);
134  const uint8_t* imageData = static_cast<const uint8_t*>(data) + sizeof(CameraFramePacket);
135 
136  // 计算图像亮度
137  int width = packet->width;
138  int height = packet->height;
139  int stride = packet->stride > 0 ? packet->stride : width * 3;
140 
141  // 采样计算平均亮度
142  float totalBrightness = 0.0f;
143  int sampleCount = 0;
144 
145  for (int y = 0; y < height; y += 4) {
146  for (int x = 0; x < width; x += 4) {
147  int idx = y * stride + x * 3;
148  if (idx + 2 < static_cast<int>(size - sizeof(CameraFramePacket))) {
149  // Y = 0.299*R + 0.587*G + 0.114*B
150  float Y = 0.299f * imageData[idx] +
151  0.587f * imageData[idx + 1] +
152  0.114f * imageData[idx + 2];
153  totalBrightness += Y;
154  sampleCount++;
155  }
156  }
157  }
158 
159  if (sampleCount > 0) {
160  float avgBrightness = totalBrightness / sampleCount;
161 
162  // 低光照检测阈值
163  const float LOW_LIGHT_THRESHOLD = 50.0f; // 0-255范围
164 
165  if (avgBrightness < LOW_LIGHT_THRESHOLD) {
166  // 检测到低光照
167  if (currentState_ != EnvironmentState::LowLight) {
168  currentState_ = EnvironmentState::LowLight;
169  confidence_ = std::min(1.0f, (LOW_LIGHT_THRESHOLD - avgBrightness) / LOW_LIGHT_THRESHOLD);
170 
171  std::cout << "[EnvironmentDetectionNode] Low light detected!"
172  << " Brightness: " << avgBrightness
173  << " Confidence: " << confidence_ << std::endl;
174 
175  publishStatus();
176  }
177  } else if (currentState_ == EnvironmentState::LowLight) {
178  // 恢复正常
179  currentState_ = EnvironmentState::Normal;
180  confidence_ = 1.0f;
181  std::cout << "[EnvironmentDetectionNode] Light condition normal" << std::endl;
182  publishStatus();
183  }
184  }

187 void EnvironmentDetectionNode::onImuData(const void* data, size_t size) {
188  if (size < sizeof(ImuSample)) return;
189 
190  const auto* imu = static_cast<const ImuSample*>(data);
191 
192  // 计算加速度模值
193  float accNorm = std::sqrt(imu->ax * imu->ax +
194  imu->ay * imu->ay +
195  imu->az * imu->az);
196 
197  // 计算角速度模值
198  float gyroNorm = std::sqrt(imu->gx * imu->gx +
199  imu->gy * imu->gy +
200  imu->gz * imu->gz);
201 
202  // 振动检测（高加速度或高角速度）
203  const float HIGH_VIBRATION_ACC_THRESHOLD = 15.0f; // m/s^2
204  const float HIGH_VIBRATION_GYRO_THRESHOLD = 5.0f; // rad/s
205 
206  static int vibrationCount = 0;
207 
208  if (accNorm > HIGH_VIBRATION_ACC_THRESHOLD || gyroNorm > HIGH_VIBRATION_GYRO_THRESHOLD) {
209  vibrationCount++;
210 
211  if (vibrationCount > 10) { // 持续振动
212  // 可能是恶劣天气（风、雨）或粗糙地形
213  std::cout << "[EnvironmentDetectionNode] High vibration detected!"
214  << " Acc: " << accNorm << " Gyro: " << gyroNorm << std::endl;
215  }
216  } else {
217  vibrationCount = std::max(0, vibrationCount - 1);
218  }
219 }

221 void EnvironmentDetectionNode::onGnssData(const void* data, size_t size) {
222  if (size < sizeof(GnssSample)) return;
223 
224  const auto* gnss = static_cast<const GnssSample*>(data);
225 
226  // GPS拒止检测
227  // 基于HDOP（水平精度因子）和卫星数量
228  const float HIGH_HDOP_THRESHOLD = 5.0f;
229  const int MIN_SATELLITES = 6;
230 
231  if (gnss->hdop > HIGH_HDOP_THRESHOLD || gnss->satellitesUsed < MIN_SATELLITES) {
232  if (currentState_ != EnvironmentState::GpsDenied) {
233  currentState_ = EnvironmentState::GpsDenied;
234 
235  // 计算置信度
236  float hdopConfidence = std::min(1.0f, (gnss->hdop - HIGH_HDOP_THRESHOLD) / 5.0f);
237  float satConfidence = std::min(1.0f,
238  static_cast<float>(MIN_SATELLITES - gnss->satellitesUsed) / MIN_SATELLITES);
239  confidence_ = std::max(hdopConfidence, satConfidence);
240 
241  std::cout << "[EnvironmentDetectionNode] GPS denied environment!"
242  << " HDOP: " << gnss->hdop
243  << " Satellites: " << gnss->satellitesUsed
244  << " Confidence: " << confidence_ << std::endl;
245 
246  publishStatus();
247  }
248  } else if (currentState_ == EnvironmentState::GpsDenied) {
249  // GPS恢复
250  currentState_ = EnvironmentState::Normal;
251  confidence_ = 1.0f;
252  std::cout << "[EnvironmentDetectionNode] GPS signal recovered" << std::endl;
253  publishStatus();
254  }
