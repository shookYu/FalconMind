27 float SimpleTrackerBackend::iouBetween(const DetectionBBox& a, const DetectionBBox& b) const {
28  // 假设 (x,y) 为左上角，width/height 为宽高
29  float ax2 = a.x + a.width;
30  float ay2 = a.y + a.height;
31  float bx2 = b.x + b.width;
32  float by2 = b.y + b.height;
33  float ix1 = std::max(a.x, b.x);
34  float iy1 = std::max(a.y, b.y);
35  float ix2 = std::min(ax2, bx2);
36  float iy2 = std::min(ay2, by2);
37  float iw = std::max(0.f, ix2 - ix1);
38  float ih = std::max(0.f, iy2 - iy1);
39  float inter = iw * ih;
40  float areaA = a.width * a.height;
41  float areaB = b.width * b.height;
42  float union_ = areaA + areaB - inter;
43  if (union_ <= 0.f) return 0.f;
44  return inter / union_;
45 }

52 bool SimpleTrackerBackend::run(DetectionResult& detections, TrackingResult& outTracks) {
53  if (!loaded_) {
54  std::cerr << "[SimpleTrackerBackend] run() called before load()" << std::endl;
55  return false;
56  }
57 
58  outTracks.frameId = detections.frameId;
59  outTracks.timestampNs = detections.timestampNs;
60  outTracks.frameIndex = detections.frameIndex;
61  outTracks.tracks.clear();
62 
63  const std::uint64_t timestampNs = detections.timestampNs;
64 
65  // 1) 对所有当前 active track 增加 missed 计数；稍后若被匹配会置 0
66  for (auto& kv : activeTracks_)
67  kv.second.missedFrames++;
68 
69  // 2) 对每个检测做 IoU 匹配：优先匹配 trackId 已存在的，否则找最大 IoU 的 active track
70  for (size_t i = 0; i < detections.detections.size(); ++i) {
71  Detection& det = detections.detections[i];
72  float bestIou = iouThreshold_;
73  int bestTrackId = -1;
74  if (det.trackId > 0 && activeTracks_.count(det.trackId)) {
75  float iou = iouBetween(activeTracks_.at(det.trackId).lastBbox, det.bbox);
76  if (iou >= iouThreshold_) {
77  bestIou = iou;
78  bestTrackId = det.trackId;
79  }
80  }
81  if (bestTrackId <= 0) {
82  for (const auto& kv : activeTracks_) {
83  if (kv.second.missedFrames > maxMissedFrames_) continue;
84  float iou = iouBetween(kv.second.lastBbox, det.bbox);
85  if (iou > bestIou) {
86  bestIou = iou;
87  bestTrackId = kv.first;
88  }
89  }
90  }
91  if (bestTrackId > 0) {
92  det.trackId = bestTrackId;
93  TrackRecord& rec = activeTracks_.at(bestTrackId);
94  rec.lastBbox = det.bbox;
95  rec.lastTimestampNs = timestampNs;
96  rec.missedFrames = 0;
97  rec.classId = det.classId;
98  rec.className = det.className;
99  rec.trajectory.push_back(TrackHistoryPoint{timestampNs, det.bbox});
100  pruneTrajectory(rec.trajectory, maxTrajectoryPoints_);
101  } else {
102  int newId = nextTrackId_++;
103  det.trackId = newId;
104  TrackRecord rec;
105  rec.trackId = newId;
106  rec.lastBbox = det.bbox;
107  rec.lastTimestampNs = timestampNs;
108  rec.missedFrames = 0;
109  rec.classId = det.classId;
110  rec.className = det.className;
111  rec.trajectory.push_back(TrackHistoryPoint{timestampNs, det.bbox});
112  activeTracks_[newId] = std::move(rec);
113  }
114  }
115 
116  // 3) 移除超过 maxMissedFrames 的 track，并写入 outTracks（ACTIVE + 本帧新 LOST）
117  std::vector<int> toRemove;
118  for (auto& kv : activeTracks_) {
119  TrackingState state;
120  state.trackId = kv.second.trackId;
121  state.targetClassId = kv.second.classId;
122  state.targetClassName = kv.second.className;
123  state.trajectory = kv.second.trajectory;
124  if (kv.second.missedFrames > maxMissedFrames_) {
125  state.status = "LOST";
126  toRemove.push_back(kv.first);
127  } else {
128  state.status = "ACTIVE";
129  }
130  outTracks.tracks.push_back(std::move(state));
131  }
132  for (int id : toRemove)
133  activeTracks_.erase(id);
134 
135  return true;
136 }
